# 2. Release-build DRM scheduler ftrace sweep (base wheel +9d6f4b188, timeout 600s per budget)
# Budgets: default (unset -> 5000), 8000, 10000, 15000, 20000, 30000, 40000, 60000, 80000, 0
# Tracepoints: gpu_scheduler:drm_sched_job_{queue,run,done} (clock=mono, buffer=65536 KB).
# As recorded: the capture used the global tracefs instance. It clears the global trace and event
# selection, sets the clock and buffer size, and leaves global tracing off afterwards.
T=/sys/kernel/tracing
sudo sh -c "echo 0 > $T/tracing_on; echo > $T/trace; echo mono > $T/trace_clock; echo 65536 > $T/buffer_size_kb; echo > $T/set_event; \
  for e in queue run done; do echo 1 > $T/events/gpu_scheduler/drm_sched_job_\$e/enable; done; echo 1 > $T/tracing_on"
# Default arm: env -u MLX_OMARCHY_BATCH_WORK ...; swept arms: env MLX_OMARCHY_BATCH_WORK=<w> ...
env MLX_OMARCHY_BATCH_WORK=<w> MLX_OMARCHY_NO_MLXLM_WARNING=1 HF_HUB_OFFLINE=1 \
  [venv-base]/bin/python -I receipts/2026-11-22-m3-pro-g15s-batch-work/tools/decode_jobs.py \
  [models]/<model> 64 [wheel-base] [checkout] > [out]/<label>.drv 2>&1
sudo sh -c "echo 0 > $T/tracing_on; cat $T/trace" > [out]/<label>.trace
sudo sh -c "echo 0 > $T/events/gpu_scheduler/enable; echo > $T/trace; echo local > $T/trace_clock"
# To reproduce without touching global tracefs, use a dedicated instance instead:
#   I=/sys/kernel/tracing/instances/batch-work
#   sudo sh -c "mkdir $I; echo mono > $I/trace_clock; echo 65536 > $I/buffer_size_kb; \
#     for e in queue run done; do echo 1 > $I/events/gpu_scheduler/drm_sched_job_\$e/enable; done; echo 1 > $I/tracing_on"
#   (run the model leg as above)
#   sudo sh -c "echo 0 > $I/tracing_on; cat $I/trace" > [out]/<label>.trace; sudo rmdir $I
# Parse only the benchmark PID (from the .drv PID line); other jobs count as foreign_jobs_in_window (0 in every window)
python3 -I receipts/2026-11-22-m3-pro-g15s-batch-work/tools/parse_ftrace.py [out]/<label>.drv [out]/<label>.trace > [out]/<label>.json

# 3. Balanced-order decode sweep (base wheel +9d6f4b188, 3 balanced rounds: fwd/rev/fwd, timeout 300s per run; default arm uses env -u MLX_OMARCHY_BATCH_WORK)
# Qwen3.8-2B 4-bit (128 tokens) and Qwen3-4B-Instruct-2507 4-bit (96 tokens)
env MLX_OMARCHY_BATCH_WORK=<w> MLX_OMARCHY_NO_MLXLM_WARNING=1 HF_HUB_OFFLINE=1 \
  [venv-base]/bin/python -I receipts/2026-11-22-m3-pro-g15s-batch-work/tools/decode_sub.py \
  [models]/<model> [out]/<tag>.json <tag> <128|96> [wheel-base] [checkout]

# 4. Balanced score_bench A/B (base wheel +9d6f4b188, default vs 15000, 2 balanced rounds, timeout 600s per run)
env MLX_OMARCHY_BATCH_WORK=15000 (or env -u MLX_OMARCHY_BATCH_WORK for the default arm) MLX_OMARCHY_NO_MLXLM_WARNING=1 HF_HUB_OFFLINE=1 \
