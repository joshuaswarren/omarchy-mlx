cmake_minimum_required(VERSION 3.25)
project(qmv_small_t_calibrate LANGUAGES CXX)
if(NOT EXISTS "${MLX_INCLUDE_DIR}/mlx/array.h" OR NOT EXISTS "${MLX_LIBRARY}")
  message(FATAL_ERROR "Exact current MLX headers/library required; no fetch or fallback")
endif()
add_executable(qmv-small-t-calibrate calibrate.cpp)
target_compile_features(qmv-small-t-calibrate PRIVATE cxx_std_20)
target_include_directories(qmv-small-t-calibrate PRIVATE "${MLX_INCLUDE_DIR}")
target_link_libraries(qmv-small-t-calibrate PRIVATE "${MLX_LIBRARY}" ${CMAKE_DL_LIBS} pthread)
get_filename_component(MLX_LIBRARY_DIR "${MLX_LIBRARY}" DIRECTORY)
set_target_properties(qmv-small-t-calibrate PROPERTIES BUILD_RPATH "${MLX_LIBRARY_DIR}")
add_executable(calibration-contract-test test_contract.cpp)
target_compile_features(calibration-contract-test PRIVATE cxx_std_20)
enable_testing()
target_compile_options(calibration-contract-test PRIVATE -UNDEBUG)
add_test(NAME calibration-contract COMMAND calibration-contract-test)
